package board

import (
	"strings"
	"testing"
)

func TestArchiveSuggestionLineEndingsProbe(t *testing.T) {
	for _, tc := range []struct {
		name   string
		runs   int
		stray  bool
		reject bool
	}{
		{"header-two", 2, false, true}, {"header-three", 3, false, true},
		{"stray-one", 1, true, false}, {"stray-two", 2, true, true},
	} {
		t.Run(tc.name, func(t *testing.T) {
			root := tempBoard(t)
			id := archiveCard(t, root, "eol-probe")
			if tc.stray {
				s := transactionSnapshot(t, root, id)
				updateSnapshot(t, root, s, s.Text+"\n## NOTE\n\none CRLF\r\nmore LF\n")
			}
			for i := 0; i < tc.runs; i++ {
				run := finalizedRun(t, root, archiveInput([]string{id}, "probe-"+itoa(i), "PM"))
				publishRun(t, root, run.RunID)
				if i == 0 && !tc.stray {
					s := transactionSnapshot(t, root, id)
					updateSnapshot(t, root, s, strings.Replace(s.Text, "## REVIEWS\n", "## REVIEWS\r\n", 1))
				}
			}
			before := transactionSnapshot(t, root, id)
			body, _ := SectionBody(before.Text, "REVIEWS")
			next := strings.ReplaceAll(before.Text, "\r\n", "\n")
			err := UpdateDocument(root, id, UpdateOptions{Document: "spec.md", Text: next, ExpectedRevision: before.Revision})
			if tc.reject {
				if err == nil || !strings.Contains(err.Error(), "REVIEWS") {
					t.Fatalf("expected managed rejection, got %v", err)
				}
				after := transactionSnapshot(t, root, id)
				if before.Text != after.Text || before.Revision != after.Revision {
					t.Fatal("rejected update changed card")
				}
			} else if err != nil {
				t.Fatal(err)
			}
			t.Logf("managed body interior CR=%d normalization rejected=%v error=%v", strings.Count(body, "\r"), err != nil, err)
			code, _, stderr, err := CheckBoard(root, []string{id}, false)
			if code != 0 || err != nil {
				t.Fatalf("%d %v %v", code, stderr, err)
			}
		})
	}
}

func TestArchiveSuggestionPublishedDoneProbe(t *testing.T) {
	root := tempBoard(t)
	id := archiveCard(t, root, "done-probe")
	run := finalizedRun(t, root, archiveInput([]string{id}, "done-run", "PM"))
	publishRun(t, root, run.RunID)
	s := transactionSnapshot(t, root, id)
	updateSnapshot(t, root, s, strings.Replace(s.Text, "## SUMMARY\n\n<FILL_IN>", "## SUMMARY\n\nCompleted fixture", 1))
	s = transactionSnapshot(t, root, id)
	if _, err := MoveWithOptions(s.Entry, root, "done", MoveOptions{Result: "completed"}); err != nil {
		t.Fatal(err)
	}
	before := transactionSnapshot(t, root, id)
	publishRun(t, root, run.RunID)
	if err := ReviewPublicationComplete(root, run.RunID); err != nil {
		t.Fatal(err)
	}
	after := transactionSnapshot(t, root, id)
	if before.Revision != after.Revision || before.Text != after.Text || after.Entry.State != "done" {
		t.Fatal("retry rewrote completed card")
	}
	_, _, err := PrepareReviewRun(root, archiveInput([]string{id}, "new-run", "PM"), archiveRequirements(), nil, archiveOriginals(), "test")
	if err == nil {
		t.Fatal("new run accepted done card")
	}
	t.Logf("published done card retry verified without writes; new intent rejected: %v", err)
}
