package notify

import (
	"os"
	"path/filepath"
	"runtime"
	"testing"

	"github.com/dualface/kander/internal/board"
)

func TestAuditPromptEchoCountsAsAcknowledgement(t *testing.T) {
	if runtime.GOOS == "windows" {
		t.Skip("POSIX terminal echo fixture")
	}
	temp := t.TempDir()
	t.Setenv("KANDER_AUDIT_ECHO", filepath.Join(temp, "echo"))
	program := filepath.Join(temp, "tmux")
	script := "#!/bin/sh\nif [ \"$1\" = \"send-keys\" ]; then\n  if [ \"$4\" = \"-l\" ]; then printf '%s' \"$5\" > \"$KANDER_AUDIT_ECHO\"; fi\n  exit 0\nfi\nif [ \"$1\" = \"capture-pane\" ]; then /bin/cat \"$KANDER_AUDIT_ECHO\"; exit 0; fi\nexit 1\n"
	if err := os.WriteFile(program, []byte(script), 0700); err != nil {
		t.Fatal(err)
	}
	marker := "KANDER-NOTIFY-ACK:audit-echo"
	instruction := notifyInstruction(board.Entry{TaskID: "20260907-audit-task"}, "/tmp/audit-message.txt", marker)
	confirmed, detail, err := tmuxDirectNotify(program, "%1", instruction, marker, .05)
	if err != nil {
		t.Fatal(err)
	}
	if !confirmed {
		t.Fatalf("confirmation: %v, detail: %s", confirmed, detail)
	}
	t.Log("confirmation succeeded using only echoed input; no executing agent acknowledgement was generated")
}
