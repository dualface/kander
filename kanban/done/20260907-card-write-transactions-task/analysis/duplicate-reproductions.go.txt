package window

import (
 "os"
 "path/filepath"
 "testing"

 "github.com/dualface/kander/internal/board"
)

func duplicateAuditBoard(t *testing.T) (string, string, string, board.Entry) {
 t.Helper()
 root := t.TempDir()
 for _, state := range board.States {
  if err := os.Mkdir(filepath.Join(root, state), 0700); err != nil { t.Fatal(err) }
 }
 id := "20260907-duplicate-audit-task"
 oldPath := filepath.Join(root, "review", id+".md")
 newPath := filepath.Join(root, "working", id+".md")
 if err := os.WriteFile(oldPath, []byte("# Original card\n"), 0600); err != nil { t.Fatal(err) }
 return root, oldPath, newPath, board.Entry{TaskID:id, State:"review", Path:oldPath, Document:oldPath, Kind:"small"}
}

func TestAuditRollbackRecreatesMovedCard(t *testing.T) {
 root, oldPath, newPath, entry := duplicateAuditBoard(t)
 if err := os.Rename(oldPath, newPath); err != nil { t.Fatal(err) }
 if err := RestoreWindowText(root, entry, "# Old text restored\n"); err != nil { t.Fatal(err) }
 for _, path := range []string{oldPath,newPath} {
  if _, err := os.Stat(path); err != nil { t.Fatal(err) }
 }
 scanned, err := board.ScanTargets(root, []string{entry.TaskID})
 if err != nil { t.Fatal(err) }
 if len(scanned.Problems) == 0 { t.Fatal("duplicate not detected") }
 t.Log("internal rollback recreated review card after it had moved to working; both copies now exist")
}

func TestAuditGuardCheckDoesNotCoverLaterMove(t *testing.T) {
 root, oldPath, newPath, entry := duplicateAuditBoard(t)
 verdict, err := board.GuardWrite(root, oldPath)
 if err != nil { t.Fatal(err) }
 if !verdict.Allowed { t.Fatal("existing card unexpectedly rejected") }
 if err := os.Rename(oldPath, newPath); err != nil { t.Fatal(err) }
 if err := os.WriteFile(oldPath, []byte("# Editor writes cached path\n"), 0600); err != nil { t.Fatal(err) }
 scanned, err := board.ScanTargets(root, []string{entry.TaskID})
 if err != nil { t.Fatal(err) }
 if len(scanned.Problems) == 0 { t.Fatal("duplicate not detected") }
 t.Log("guard allowed existing path; move between guard and actual write still produced duplicate")
}
